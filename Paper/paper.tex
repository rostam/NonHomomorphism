\begin{abstract}\sloppy
The \emph{non-homomorphism factor} $|G,H|$ is the minimum number of edges that
must be removed from $G$ so that the remaining graph admits a homomorphism to
$H$.  The \emph{concentration parameter} $\gamma(G,H)$ is the minimum, over all
homomorphisms $f\colon G\to H$, of the largest number of edges of $G$ that $f$
sends onto a single edge of $H$.

For $|G,H|$ we prove $|K_n,K_m|=\binom n2-t(n,m)$ and
$|K_n,H|=\binom n2-\lfloor n^2/4\rfloor$ for every triangle-free $H$ with an
edge.  For a $d$-regular graph $G$ on $n$ vertices and triangle-free $H$ we prove
$|L(G),H|\ge n\lfloor (d-1)^2/4\rfloor$, with equality when $G$ has a balanced
$2$-edge-colouring.  We prove the quantitative Hedetniemi bound
\[|G\times H,K|\le\min(2|E(H)|\,|G,K|,\,2|E(G)|\,|H,K|)\]
and the exact value
$|C_p\times C_q,K_2|=2p$ for odd $p\le q$.  For cubic $G$ we show that
$|L(G),L(P)|$, where $P$ is the Petersen graph, is at least the number of leaf
blocks of the bridge-block tree of $G$.

For $\gamma$ we prove multiplicativity, monotonicity, and the transfer bound
$|G,K|\le\gamma(G,H)\,|H,K|$.  We find an exact identity for the loads of a
homomorphism to $K_3$ in terms of the degree sums of its colour classes.  It
gives exact values of $\gamma(\cdot,K_3)$ for cycles, complete bipartite graphs,
the Petersen graph, books, friendship graphs, wheels and complete tripartite
graphs.  We also show that $\max(|G,H|,|H,G|)$ is not a pseudo-metric.

Every result in this paper is formally verified in the Lean~4 proof assistant
with the Mathlib library.
\end{abstract}

\section{Introduction}
% ─────────────────────────────────────────────────────────────────────────────

All graphs are finite, simple and undirected.

\begin{definition}[Graph homomorphism]
A \emph{homomorphism} from $G$ to $H$ is a map $f\colon V(G)\to V(H)$ such that
$\{u,v\}\in E(G)$ implies $\{f(u),f(v)\}\in E(H)$.  We write $G\to H$ when such
a map exists, and $G\leftrightarrow H$ when $G\to H$ and $H\to G$.
\end{definition}

A proper $k$-colouring of $G$ is the same as a homomorphism $G\to K_k$.  Two
standard facts~\cite{hell2004} follow directly.

\begin{proposition}
\label{prop:chi_mono}
If $G\to H$ then $\chi(G)\le\chi(H)$.
\end{proposition}

\begin{proposition}
\label{prop:hom_contracts}
Every homomorphism $f\colon G\to H$ satisfies $d_H(f(u),f(v))\le d_G(u,v)$ for
all $u,v\in V(G)$.
\end{proposition}

\begin{proof}
A walk of length $\ell$ from $u$ to $v$ maps to a walk of length $\ell$ from
$f(u)$ to $f(v)$.
\end{proof}

\paragraph{Formal verification.}
Every theorem, lemma, proposition and corollary in this paper has been proved in
Lean~4~\cite{lean4} using Mathlib~\cite{mathlib}.  The only exceptions are the
two classical results quoted in Remark~\ref{rem:balanced}.  The development is in
the \texttt{lean/NonHom} directory of the repository.
Section~\ref{sec:formal} describes it and lists the Lean name of each result.

\section{Non-Homomorphism Factors}
% ─────────────────────────────────────────────────────────────────────────────

\begin{definition}[Non-homomorphism factor~\cite{khoshkha2005}]
\label{def:nhf}
$|G,H|$ is the minimum number of edges that must be removed from $G$ to obtain a
graph $G'$ with $G'\to H$.  In particular $|G,H|=0$ if and only if $G\to H$.
\end{definition}

\begin{remark}[Complexity]
For fixed $H$, deciding $|G,H|\le k$ is \textsc{Edge Deletion-$H$-Colouring},
studied by Foucaud, Hocquard, Lajou, Mitsou and Pierron~\cite{foucaud2019}.  The
case $k=0$ is \textsc{$H$-Colouring}.  So by the Hell--Ne\v{s}et\v{r}il
dichotomy~\cite{hell2004}, computing $|G,H|$ is NP-hard whenever the core of $H$
has more than one edge.  For $H=K_2$ the problem is \textsc{Edge Bipartization},
which is NP-complete but fixed-parameter tractable in $k$.
\end{remark}

\begin{proposition}[\cite{khoshkha2005}]
\label{prop:monotone}
\begin{enumerate}
  \item\label{it:mon1} If $H_1\to H_2$ then $|G,H_1|\ge|G,H_2|$ for every $G$.
  \item\label{it:mon2} If $H_1\leftrightarrow H_2$ then $|G,H_1|=|G,H_2|$ for every $G$.
  \item\label{it:mon3} If $f\colon G_1\to G_2$ is edge-injective (distinct edges
        map to distinct edges) then $|G_1,H|\le|G_2,H|$ for every $H$.
\end{enumerate}
\end{proposition}

\begin{proof}
(\ref{it:mon1}) If $G'\to H_1$ then $G'\to H_1\to H_2$.
(\ref{it:mon2}) Apply (\ref{it:mon1}) in both directions.
(\ref{it:mon3}) Take $S\subseteq E(G_2)$ with $|S|=|G_2,H|$ and $G_2-S\to H$, and
let $S'=\{e\in E(G_1):f(e)\in S\}$.  Edge-injectivity gives $|S'|\le|S|$, and $f$
restricts to a homomorphism $G_1-S'\to G_2-S$.
\end{proof}

\subsection{Complete graphs and Tur\'an's theorem}

Let $t(n,m)$ be the number of edges of the Tur\'an graph $T(n,m)$, the complete
$m$-partite graph on $n$ vertices with parts as equal as possible.  By Tur\'an's
theorem, every $K_{m+1}$-free graph on $n$ vertices has at most $t(n,m)$ edges.

\begin{theorem}
\label{thm:KnKm}
For $m\ge1$, $\displaystyle|K_n,K_m| = \binom{n}{2} - t(n,m)$.
If $n=qm+r$ with $0\le r<m$, this equals
$r\dbinom{q+1}{2}+(m-r)\dbinom{q}{2}$.
In particular $|K_{n+1},K_n|=1$ for all $n\ge1$.
\end{theorem}

\begin{proof}
Removing the edges outside $T(n,m)$ leaves an $m$-colourable graph.  Conversely,
an $m$-colourable spanning subgraph of $K_n$ is $K_{m+1}$-free, so by Tur\'an's
theorem it has at most $t(n,m)$ edges.  The explicit formula follows by induction
on $q$ from $t(n+m,m)=t(n,m)+n(m-1)+\binom m2$.  For $n+1=1\cdot n+1$ it gives
$\binom22=1$.
\end{proof}

\begin{theorem}
\label{thm:lower_bound}
For all graphs $G$ and $H$,
$|G,H|\ge \max\bigl(0,\,|E(G)|-t(|V(G)|,\chi(H))\bigr)$.
\end{theorem}

\begin{proof}
If $G'\subseteq G$ and $G'\to H$ then $\chi(G')\le\chi(H)$, so $G'$ is
$K_{\chi(H)+1}$-free and has at most $t(|V(G)|,\chi(H))$ edges.
\end{proof}

Triangle-free targets all give the same answer for complete graphs.

\begin{theorem}
\label{thm:Kn_trianglefree}
If $H$ is triangle-free and has at least one edge, then
$|K_n,H|=\binom n2-\lfloor n^2/4\rfloor$.
\end{theorem}

\begin{proof}
If $G'\to H$ then $G'$ is triangle-free, because a homomorphism maps a triangle
onto a triangle.  So by Mantel's theorem $|E(G')|\le\lfloor n^2/4\rfloor$.
Conversely $K_n-E(\overline{T(n,2)})=T(n,2)\to K_2\to H$.
\end{proof}

\subsection{Cycles and the Petersen graph}

Every cycle is $3$-colourable, so $|C_n,K_3|=0$.  For $K_2$, $|C_n,K_2|=0$ if
$n$ is even and $|C_n,K_2|=1$ if $n\ge3$ is odd.  Indeed an odd cycle is not
bipartite, and deleting one edge leaves a path.

Let $P$ be the Petersen graph, with outer cycle $u_0u_1u_2u_3u_4$, spokes
$u_iv_i$, and inner edges $v_iv_{i+2}$ (indices mod $5$).  It has $10$ vertices,
$15$ edges, and $\chi(P)=3$, so $|P,K_m|=0$ for $m\ge3$.

\begin{theorem}
\label{thm:PK2}
$|P,K_2|=3$.
\end{theorem}

\begin{proof}
$P$ has exactly twelve $5$-cycles, and each edge lies on exactly four of them.
Two edges therefore lie on at most eight, so after deleting any two edges some
$5$-cycle survives and the graph is not bipartite.  Hence $|P,K_2|\ge3$.
Conversely, deleting $u_0u_1$, $u_3v_3$ and $v_2v_4$ leaves a bipartite graph
with sides $\{u_0,u_1,u_3,v_2,v_3,v_4\}$ and $\{u_2,u_4,v_0,v_1\}$.
\end{proof}

Theorem~\ref{thm:lower_bound} only gives $|P,K_2|\ge15-t(10,2)<0$ here, so the
odd-cycle structure, not edge density, is what forces the deletions.

\subsection{Line graphs of regular graphs}

\begin{theorem}
\label{thm:line}
Let $G$ be $d$-regular on $n$ vertices and let $H$ be triangle-free.  Then
\[
  |L(G),H|\;\ge\; n\left(\binom d2-\left\lfloor\frac{d^2}{4}\right\rfloor\right)
  \;=\; n\left\lfloor\frac{(d-1)^2}{4}\right\rfloor .
\]
Suppose moreover that $H$ has an edge and that $E(G)$ has a $2$-colouring that
is \emph{balanced}, meaning that at every vertex the two colours occur
$\lfloor d/2\rfloor$ and $\lceil d/2\rceil$ times.  Then equality holds.
\end{theorem}

\begin{proof}
The $d$ edges at a vertex $v$ form a clique $Q_v\cong K_d$ in $L(G)$.  Distinct
cliques share no edge of $L(G)$, since two distinct edges of $G$ share at most one
vertex.  If $L(G)-S\to H$ then $L(G)-S$ is triangle-free.  By Mantel's theorem
it keeps at most $\lfloor d^2/4\rfloor$ of the $\binom d2$ edges of each $Q_v$.
Summing over $v$ gives the lower bound.

For the upper bound, colour the vertices of $L(G)$ by the given edge colouring,
and delete the edges of $L(G)$ whose ends have the same colour.  What remains maps
to $K_2\to H$.  At a vertex with colour counts $a$ and $d-a$ the deleted edges
number $\binom a2+\binom{d-a}2=\binom d2-a(d-a)$.  This equals
$\binom d2-\lfloor d^2/4\rfloor$ when the colouring is balanced.
\end{proof}

\begin{remark}
\label{rem:balanced}
Balanced $2$-edge-colourings exist in three cases.  If $4\mid d$, split a
$2$-factorization (Petersen's $2$-factor theorem~\cite{petersen1891}) evenly.  If
$G$ is bipartite, group the $d$ colour classes of a proper edge colouring
(K\H{o}nig~\cite{konig1916}).  If $d$ is odd and $G$ has a perfect matching $M$,
split $G-M$ into $(d-1)/2$ two-factors.  Then give colour $1$ to $M$ together with
$\lfloor (d-1)/4\rfloor$ of these two-factors.

In particular every bridgeless cubic graph $G$ on $n$ vertices has a perfect
matching, by Petersen's theorem~\cite{petersen1891}.  So $|L(G),H|=n$ for every
triangle-free $H$ with an edge, and the whole ladder
$|L(G),C_5|=|L(G),C_7|=\dots=|L(G),K_2|$ collapses.

These two classical theorems are not formalized.  The Lean proof establishes the
cubic case for every cubic graph that has a perfect matching.
\end{remark}

\begin{theorem}
\label{thm:square_cycle}
Let $C_n^2$ ($n\ge5$) be the circulant graph on $\mathbb Z_n$ with jumps $1$ and
$2$.  For every triangle-free $H$, $|C_n^2,H|\ge\lceil n/2\rceil$.
\end{theorem}

\begin{proof}
The $n$ triangles $\{i,i+1,i+2\}$ must each lose an edge.  An edge
$\{i,i+1\}$ lies only in the triangles starting at $i-1$ and at $i$, and an edge
$\{i,i+2\}$ lies only in the triangle starting at $i$.  So each deleted edge
destroys at most two of these triangles.
\end{proof}

\subsection{Tensor products}

The tensor product $G\times H$ has vertex set $V(G)\times V(H)$, with
$(g,h)\sim(g',h')$ if and only if $g\sim g'$ and $h\sim h'$.

\begin{theorem}[Quantitative Hedetniemi bound]
\label{thm:hedetniemi}
For all graphs $G$, $H$ and every graph $K$ with at least one vertex,
\[
  |G\times H,K|\;\le\;\min\bigl(2|E(H)|\cdot|G,K|,\;2|E(G)|\cdot|H,K|\bigr).
\]
\end{theorem}

\begin{proof}
The projection $G\times H\to G$ sends exactly $2|E(H)|$ edges onto each edge of
$G$, one for each orientation of each edge of $H$.  Now apply
Lemma~\ref{lem:nhf_bound} to this map.  If $G\times H$ has no edges it maps to
any graph with a vertex.  The second bound is symmetric.
\end{proof}

The hypothesis on $K$ is needed.  If $K$ has no vertices and $H$ has no edges,
the right-hand side is $0$ but the left-hand side is infinite.  By
Shitov's counterexample to Hedetniemi's conjecture~\cite{shitov2019}, the bound
can be far from tight for large targets: the left-hand side can be $0$ while
neither factor is $k$-colourable.  For $K=K_2$ and two odd cycles it is tight.

\begin{theorem}
\label{thm:CpCq}
For odd $3\le p\le q$, $|C_p\times C_q,K_2|=2p$.
\end{theorem}

\begin{proof}
The upper bound is Theorem~\ref{thm:hedetniemi}, since $|C_q,K_2|=1$.

For the lower bound, let $\varepsilon_i=+1$ for $0\le i<(q+p)/2$ and
$\varepsilon_i=-1$ for $(q+p)/2\le i<q$, and let $g_i=\sum_{j<i}\varepsilon_j$.
Then $g_q=p\equiv0\pmod p$.  For $a\in\mathbb Z_p$, $h\in\mathbb Z_q$ and
$\sigma\in\{\pm1\}$, the walk $i\mapsto(a+\sigma g_i,\,h+i)$, $0\le i\le q$, is a
closed walk of odd length $q$ in $C_p\times C_q$.  So each of these $2pq$ walks
must contain a deleted edge.

Fix a step index $i$ and an edge $e$.  The second coordinate increases by $1$
along every step, so it determines which end of $e$ the step leaves from.  The
first coordinate then determines $\sigma\varepsilon_i$, hence $\sigma$, and then
$a$ and $h$.  So at most one walk uses $e$ at step $i$.  Charging each walk to
the step at which it meets the deletion set $S$ gives $2pq\le q|S|$.
\end{proof}

\subsection{Petersen colourings}

A \emph{Petersen colouring} of a cubic graph $G$ is a map $E(G)\to E(P)$ that
sends the three edges at each vertex onto the three edges at some vertex of $P$.
Since $P$ is triangle-free, every triangle of $L(P)$ is the star of a vertex.  So
Petersen colourings of $G$ are exactly the homomorphisms $L(G)\to L(P)$, and
Jaeger's Petersen colouring conjecture~\cite{jaeger1988} says that
$|L(G),L(P)|=0$ for every bridgeless cubic $G$.

\begin{theorem}
\label{thm:jaeger}
Let $G$ be cubic.  Let $X_1,\dots,X_t\subseteq V(G)$ be pairwise disjoint sets,
each of which is left by exactly one edge of $G$.  Then $|L(G),L(P)|\ge t$.  In
particular $|L(G),L(P)|$ is at least the number of leaves of the bridge-block tree
of $G$.
\end{theorem}

\begin{proof}
Suppose $\varphi\colon L(G)-S\to L(P)$.  Call $x\in V(G)$ \emph{good} if none of
the three edges of $L(G)$ between edges at $x$ lies in $S$.  For a good $x$, the
images of the three edges at $x$ form a triangle of $L(P)$, which is a star.

$P$ has exactly six perfect matchings, and each edge lies in exactly two of them.
Fix $X=X_i$, its leaving edge $b$, and a perfect matching $M$.  Each good $x$ has
exactly one incident edge whose image is in $M$.  If every vertex of $X$ is good,
counting incidences gives $|X|=2m+[\varphi(b)\in M]$, where $m$ is the number of
edges inside $X$ with image in $M$.  But $3|X|=2|E(X)|+1$, so $|X|$ is odd and
$\varphi(b)\in M$.  This holds for all six matchings, which is impossible.

So each $X_i$ contains a bad vertex $x_i$, together with an edge
$\{e_i,f_i\}\in S$ of $L(G)$ where $e_i,f_i$ are distinct edges at $x_i$.  Two
distinct edges of $G$ share at most one vertex, so these $t$ elements of $S$ are
distinct.  The vertex sets of the leaf blocks of the bridge-block tree satisfy
the hypothesis, each being left only by its bridge.
\end{proof}

\section{The Concentration Parameter}
% ─────────────────────────────────────────────────────────────────────────────

\begin{definition}[Concentration parameter]
\label{def:gamma}
For a homomorphism $f\colon G\to H$ and $e\in E(H)$, the \emph{load} $\mu(f,e)$
is the number of edges of $G$ mapped onto $e$.  Let
$\mu(f)=\max_{e}\mu(f,e)$ and
\[
  \gamma(G,H) = \min_{f\colon G\to H} \mu(f),
\]
with $\gamma(G,H)=+\infty$ when $G\not\to H$.
\end{definition}

A related per-homomorphism quantity $M^\sigma$ appears in~\cite{daneshgar2007}.

\begin{proposition}[Pigeonhole bound]
\label{prop:pigeonhole}
If $G\to H$ then $|E(G)|\le|E(H)|\cdot\gamma(G,H)$, so
$\gamma(G,H)\ge\lceil|E(G)|/|E(H)|\rceil$.
\end{proposition}

\begin{proof}
The loads of $f$ sum to $|E(G)|$.
\end{proof}

\begin{lemma}
\label{lem:gamma_basic}
If $|E(G)|\ge1$ and $G\to H$ then $\gamma(G,H)\ge1$.  Moreover $\gamma(G,G)=1$.
\end{lemma}

\begin{lemma}[Multiplicativity]
\label{lem:gamma_mult}
If $G\to H\to K$ then $\gamma(G,K)\le\gamma(G,H)\cdot\gamma(H,K)$.
\end{lemma}

\begin{proof}
Take optimal $f\colon G\to H$ and $g\colon H\to K$.  An edge $e$ of $K$ has at
most $\mu(g)$ preimage edges in $H$, and each of these has at most $\mu(f)$
preimages in $G$.  So $\mu(g\circ f)\le\mu(f)\mu(g)$.
\end{proof}

\begin{lemma}[Transfer bound]
\label{lem:nhf_bound}
If $G\to H$ and $|E(G)|\ge1$, then $|G,K|\le\gamma(G,H)\cdot|H,K|$ for every $K$.
More generally $|G,K|\le\mu(f)\,|H,K|$ for every homomorphism $f\colon G\to H$.
\end{lemma}

\begin{proof}
Let $H-S\to K$ with $|S|=|H,K|$, and let $T$ be the set of edges of $G$ that $f$
maps into $S$.  Then $|T|\le\mu(f)|S|$, and $f$ restricts to
$G-T\to H-S\to K$.
\end{proof}

\begin{corollary}[Two-sided bound]
\label{cor:two_sided}
If $G\to H$ and $K\to H$, and both have an edge, then
$\max(|G,K|,|K,G|)\le\max\bigl(\gamma(G,H)|H,K|,\ \gamma(K,H)|H,G|\bigr)$.
\end{corollary}

\begin{lemma}[Monotonicity in the target]
\label{lem:gamma_mono}
If $H_1$ embeds injectively in $H_2$ (for instance $H_1\subseteq H_2$), then
$\gamma(G,H_2)\le\gamma(G,H_1)$.
\end{lemma}

\begin{proof}
Composing with an injective homomorphism does not change any load.
\end{proof}

\begin{proposition}
\label{prop:gamma_Kn}
For $2\le n\le m$, $\gamma(K_n,K_m)=1$.  For $n>m$, $\gamma(K_n,K_m)=+\infty$.
\end{proposition}

\subsection{An exact identity for homomorphisms to $K_3$}

For $f\colon G\to K_3$ with colour classes $A_0,A_1,A_2$, let
$D_i=\sum_{v\in A_i}\deg v$ and write $L_{ij}=\mu(f,\{i,j\})$.

\begin{lemma}
\label{lem:degree_identity}
For every graph $G$, every $f\colon G\to K_3$ and distinct colours $a,b,c$,
\[
  2L_{ab}=D_a+D_b-D_c,
  \qquad\text{hence}\qquad
  2\mu(f)=\max(D_0+D_1-D_2,\;D_0+D_2-D_1,\;D_1+D_2-D_0).
\]
\end{lemma}

\begin{proof}
Each edge at a vertex of $A_a$ goes to $A_b$ or to $A_c$, so $D_a=L_{ab}+L_{ac}$.
Solve the three equations.
\end{proof}

\begin{theorem}[$d$-regularity formula]
\label{thm:d_regular}
Let $G$ be $d$-regular on $n$ vertices with $d\ge1$, and let $f\colon G\to K_3$
have colour classes of sizes $a,b,c$.  Then
\[
  2\mu(f) = d\cdot\max(a+b-c,\; a+c-b,\; b+c-a).
\]
The triple $(a+b-c,a+c-b,b+c-a)$ is non-negative, sums to $n$, and has constant
parity.  Hence, if $M$ is the least maximum of such a triple, then
$\gamma(G,K_3)\ge\frac d2M$.  Equality holds as soon as some proper $3$-colouring
attains $M$.
\end{theorem}

\begin{proof}
Put $D_i=d|A_i|$ in Lemma~\ref{lem:degree_identity}.  Non-negativity follows from
$d\,(a+b-c)=2L_{ab}\ge0$, and the entries differ by even numbers.
\end{proof}

\begin{theorem}
\label{thm:cycle_gamma}
For all $n\ge3$,
\[
  \gamma(C_n, K_3) =
  \begin{cases}
    \lceil n/3 \rceil     & \text{if } n \not\equiv 2 \pmod{3}, \\
    \lceil n/3 \rceil + 1 & \text{if } n \equiv 2 \pmod{3}.
  \end{cases}
\]
In particular $\gamma(C_n,K_3)$ attains the pigeonhole bound if and only if
$n\not\equiv2\pmod 3$.
\end{theorem}

\begin{proof}
Apply Theorem~\ref{thm:d_regular} with $d=2$ and write $n=3q+r$.  A same-parity
triple summing to $n$ has maximum at least $\lceil n/3\rceil$.  For $r=2$ the
only triple with maximum $q+1$ is $(q+1,q+1,q)$, which has mixed parity, so the
maximum is at least $q+2$.

To see that these bounds are attained, colour vertex $i<3q$ by $i\bmod3$.  If
$r=1$, give the last vertex colour~$1$.  If $r=2$, give the last two vertices
colours $0$ and $1$.  The class sizes are $(q,q,q)$, $(q,q+1,q)$ and
$(q+1,q+1,q)$, with maxima $q$, $q+1$ and $q+2$.
\end{proof}

\begin{theorem}
\label{thm:Knn}
$|K_{n,n},K_3|=0$ and $\gamma(K_{n,n},K_3)=n\lceil n/2\rceil$.
\end{theorem}

\begin{proof}
Let $f\colon K_{n,n}\to K_3$.  The colour sets of the two sides are disjoint, so
one side is monochromatic in a colour that the other side does not use.  That
colour class has exactly $n$ vertices.  With $d=n$ and class sizes
$(x,n-x,n)$, Theorem~\ref{thm:d_regular} gives $\mu(f)=n\max(x,n-x)\ge
n\lceil n/2\rceil$.  Splitting one side as evenly as possible attains this.
\end{proof}

For $n=1,2,3,4$ this gives $1,2,6,8$.

\begin{theorem}
\label{thm:petersen_gamma}
$\gamma(P,K_3)=6$ and $\gamma(P,K_5)=2$.
\end{theorem}

\begin{proof}
For $K_3$ take $d=3$ and $n=10$.  Same-parity triples summing to $10$ are all
even, so their maximum is at least $4$, and $\gamma(P,K_3)\ge6$.  Equality holds
because $P$ has a proper $3$-colouring with class sizes $(3,4,3)$, whose triple is
$(4,2,4)$.  For $K_5$, the pigeonhole bound $\lceil15/10\rceil=2$ is attained by
an explicit $5$-colouring.
\end{proof}

So $\gamma(P,K_3)=6$ exceeds the pigeonhole bound $5$ because of parity.  In
contrast $\gamma(P,K_5)$ attains it.

\subsection{Irregular source graphs}

The \emph{book} $B_k$ is $k$ triangles sharing one edge (the spine).  The
\emph{friendship graph} $F_k$ is $k$ triangles sharing one vertex.  The
\emph{wheel} $W_n$ is a vertex (the hub) joined to every vertex of $C_{n-1}$.

\begin{theorem}
\label{thm:families}
\begin{enumerate}
  \item For $k\ge1$: $\gamma(B_k,K_3)=k$ and $|B_k,K_2|=1$.
  \item For $k\ge0$: $\gamma(F_k,K_3)=k$ and $|F_k,K_2|=k$.
  \item For odd $n\ge5$: $\gamma(W_n,K_3)=n-1$.
  \item For $p,q,r\ge1$: $\gamma(K_{p,q,r},K_3)=\max(pq,pr,qr)$.
\end{enumerate}
\end{theorem}

\begin{proof}
All four use Lemma~\ref{lem:degree_identity}.

(1) The two spine vertices get distinct colours $a,b$, and every page vertex gets
the third colour $c$.  So $D_a=D_b=k+1$ and $D_c=2k$, giving $L_{ab}=1$ and
$L_{ac}=L_{bc}=k$.  Deleting the spine leaves a bipartite graph, and $B_k$
contains a triangle, so $|B_k,K_2|=1$.

(2) The hub gets some colour $c$.  Each triangle puts one vertex in each of the
other two classes, so $D_0=D_1=D_2=2k$ and every load equals $k$, for every
homomorphism.  The $k$ triangles are edge-disjoint, so $|F_k,K_2|\ge k$.
Deleting the $k$ edges opposite the hub leaves a star.

(3) The hub, of colour $c$, is alone in its class, so $D_c=n-1$.  The total degree
is $4(n-1)$, so $D_a+D_b=3(n-1)$ and $L_{ab}=n-1$.  Alternating two colours on
the even rim attains this.

(4) If all parts are non-empty, each part is monochromatic and the three parts
get distinct colours.  Otherwise some vertex would see three colours among its
neighbours.  The class of part $i$ has $D=s_i(N-s_i)$, where $N=p+q+r$, so
$2L=s_i(N-s_i)+s_j(N-s_j)-s_k(N-s_k)=2s_is_j$.
\end{proof}

In particular $\gamma(F_k,K_3)$ attains the pigeonhole bound $k$.  On the other
hand $\gamma(W_n,K_3)=n-1$ exceeds it for $n\ge5$, and so do
$\gamma(K_{1,2,3},K_3)=6>4$ and $\gamma(K_{2,3,4},K_3)=12>9$.  The book graphs
separate the two parameters: $|B_k,K_2|=1$ for every $k$, while
$\gamma(B_k,K_3)=k$ grows without bound.

\subsection{Cycles into odd cycles}

\begin{theorem}
\label{thm:cycle_hom}
Let $m\ge3$ be odd and $n\ge3$.  Then $C_n\to C_m$ if and only if $n$ is even or
$n\ge m$.
\end{theorem}

\begin{proof}
An even cycle maps onto an edge.  If $n\ge m$ are both odd, wind once around
$C_m$ and then oscillate on an edge for the remaining even number of steps.
Conversely, let $n<m$ with $n$ odd and $f\colon C_n\to C_m$.  Then $f$ misses some
vertex $w$, and $C_m-w$ is a path.  So $C_n$ would be bipartite, a contradiction.
\end{proof}

When $5\nmid n$ the pigeonhole bound $\lceil n/5\rceil$ can still be attained.
For example $\gamma(C_6,C_5)=2$, via $0,1,2,3,2,1$.  Similarly
$\gamma(C_8,C_7)=\gamma(C_{10},C_7)=\gamma(C_{12},C_7)=2$ and
$\gamma(C_{15},C_7)=3$ all equal $\lceil n/7\rceil$.

% ─────────────────────────────────────────────────────────────────────────────
\section{The Symmetric Combination $m(G,H)$}
\label{sec:metric}
% ─────────────────────────────────────────────────────────────────────────────

Let $m(G,H)=\max(|G,H|,|H,G|)$.  It is non-negative and symmetric, and
$m(G,G)=0$.

\begin{proposition}
\label{prop:m_zero}
$m(G,H)=0$ if and only if $G\leftrightarrow H$.
\end{proposition}

\begin{theorem}
\label{thm:no_triangle}
$m$ does not satisfy the triangle inequality, and so is not a pseudo-metric.
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:KnKm}, and since $K_n\to K_m$ for $n\le m$,
\[
  m(K_7,K_3)=5,\qquad m(K_7,K_5)=2,\qquad m(K_5,K_3)=2,
\]
and $5>2+2$.
\end{proof}

\begin{remark}
Khoshkha~\cite{khoshkha2005} claims that $m$ is a pseudo-metric.
Theorem~\ref{thm:no_triangle} shows that this is false under
Definition~\ref{def:nhf}.
\end{remark}

4.35.0.  It consists of 17 files and about 3\,300 lines.  It defines $|G,H|$ as
the infimum, in $\mathbb N\cup\{\infty\}$, of $|S|$ over finite edge sets $S$ such
that $G-S\to H$.  It defines $\gamma(G,H)$ as the infimum of $\mu(f)$ over all

\section{Open Questions}
\label{sec:open}
% ─────────────────────────────────────────────────────────────────────────────

\begin{enumerate}
\item \textbf{General targets.}  Lemma~\ref{lem:degree_identity} determines
$\mu(f)$ for $f\colon G\to K_3$ from the degree sums of the colour classes.  Is
there an analogous identity for targets $H$ other than $K_3$?

\item \textbf{When is the pigeonhole bound attained?}  For cycles into $K_3$ it
is attained exactly when $n\not\equiv2\pmod3$ (Theorem~\ref{thm:cycle_gamma}).
For $d$-regular $G$, Theorem~\ref{thm:d_regular} turns the question into one about
realisable colour-class sizes.  Characterise the pairs $(G,H)$ for which
$\gamma(G,H)=\lceil|E(G)|/|E(H)|\rceil$.

\item \textbf{Repairing $m$.}  Does the edge-edit distance to a graph mapping to
$H$, or $m$ restricted to graphs with the same number of vertices, satisfy the
triangle inequality?

\item \textbf{Petersen colourings.}  Is $|L(G),L(P)|$ equal to the number of leaf
blocks of the bridge-block tree for every cubic $G$?  The case of bridgeless $G$
is Jaeger's conjecture.

\item \textbf{Tightness of Theorem~\ref{thm:hedetniemi}.}  For which $G$, $H$ is
the bound tight for $K=K_2$ or $K=K_3$?  Theorem~\ref{thm:CpCq} gives an
infinite tight family for $K_2$.  For large targets the bound fails to be tight,
by~\cite{shitov2019}.
\end{enumerate}
